# ============================================================================
# CALENDAR ALIGNMENT FILTERING
# ============================================================================
# Problem: When a market is closed on a holiday, Bloomberg carries forward the
# previous day's price. pct_change() produces 0% for the closed market while
# open markets have real returns. PCA interprets this mismatch as a genuine
# residual, and np.cumsum() permanently accumulates it — causing extreme
# z-score swings that don't reflect real market relationships.
#
# Solution: Two-layer filtering:
#   Layer 1 (Database): Known holidays with per-market impact (closed/early_close/normal)
#   Layer 2 (Safety Net): Detect any remaining stale 0% returns automatically
# ============================================================================

def compute_easter_date(year):
    """Meeus/Jones/Butcher algorithm for Easter Sunday. Pure arithmetic, no packages."""
    a = year % 19
    b = year // 100
    c = year % 100
    d = b // 4
    e = b % 4
    f = (b + 8) // 25
    g = (b - f + 1) // 3
    h = (19 * a + b - d - g + 15) % 30
    i = c // 4
    k = c % 4
    l = (32 + 2 * e + 2 * i - h - k) % 7
    m = (a + 11 * h + 22 * l) // 451
    month = (h + l - 7 * m + 114) // 31
    day = ((h + l - 7 * m + 114) % 31) + 1
    return datetime(year, month, day).date()

def _nth_weekday(year, month, weekday, n):
    """Get the nth occurrence of a weekday in a month.
    weekday: 0=Monday, 6=Sunday. n: 1=first, 2=second, etc."""
    first_day = datetime(year, month, 1)
    offset = (weekday - first_day.weekday()) % 7
    return (first_day + timedelta(days=offset + 7 * (n - 1))).date()

def _last_weekday(year, month, weekday):
    """Get the last occurrence of a weekday in a month."""
    if month == 12:
        next_month = datetime(year + 1, 1, 1)
    else:
        next_month = datetime(year, month + 1, 1)
    last_day = next_month - timedelta(days=1)
    offset = (last_day.weekday() - weekday) % 7
    return (last_day - timedelta(days=offset)).date()

def _adjust_weekend(d):
    """US observed-holiday rule: Saturday → Friday, Sunday → Monday."""
    wd = d.weekday()
    if wd == 5:  # Saturday → observed Friday
        return d - timedelta(days=1)
    elif wd == 6:  # Sunday → observed Monday
        return d + timedelta(days=1)
    return d

def generate_recurring_holidays(start_year, end_year):
    """Auto-generate predictable recurring market holidays for each year.
    Returns list of dicts in the same format as MARKET_HOLIDAYS.

    Each entry captures the impact on credit and equity markets separately:
      "closed"      = market fully closed, Bloomberg carries forward stale price
      "early_close" = shortened session, partial real return
      "normal"      = market open as usual

    for year in range(start_year, end_year + 1):
        easter = compute_easter_date(year)

def get_dates_to_filter(holidays_db, selected_assets):
    """Determine which dates to remove based on selected assets and holiday calendar.

    Removal logic (asset-aware):
    - Remove if ANY selected asset was "closed" AND another was "normal" or "early_close"
    - Remove if ANY selected asset had "early_close" AND another had "normal"
    - Keep if ALL selected assets share the same status (no mismatch to contaminate PCA)

    If user only selects US assets, EU holidays won't cause removals (no mismatch).
    """
    # Group holidays by date
    date_holidays = {}
    for h in holidays_db:
        d = h["date"]
        if d not in date_holidays:
            date_holidays[d] = []
        date_holidays[d].append(h)

    dates_to_filter = []
    audit_rows = []

    for date_str, holiday_entries in date_holidays.items():
        # For each selected asset, determine its market status on this date
        asset_statuses = {}
        for asset in selected_assets:
            info = ASSET_MARKET_MAP.get(asset)
            if info is None:
                continue
            region = info["region"]
            asset_type = info["type"]

            # Find the holiday entry matching this asset's region
            status = "normal"
            for h in holiday_entries:
                if h["region"] == region:
                    status = h[asset_type]
                    break
            asset_statuses[asset] = status

        if not asset_statuses:
            continue

        # Check for mismatch across selected assets
        unique_statuses = set(asset_statuses.values())

        # All same status → no mismatch → keep
        if len(unique_statuses) == 1:
            continue

        # Mixed statuses → check if the mismatch contaminates PCA
        should_filter = False
        if "closed" in unique_statuses and ("normal" in unique_statuses or "early_close" in unique_statuses):
            should_filter = True
        elif "early_close" in unique_statuses and "normal" in unique_statuses:
            should_filter = True

        if should_filter:
            dt = pd.Timestamp(date_str)
            dates_to_filter.append(dt)
            regions = set(h["region"] for h in holiday_entries)
            rules = ", ".join(set(h["rule"] for h in holiday_entries))
            credit_impacts = set(h["credit"] for h in holiday_entries)
            equity_impacts = set(h["equity"] for h in holiday_entries)
            audit_rows.append({
                "Date": dt,
                "Day of Week": dt.strftime("%A"),
                "Region": "/".join(sorted(regions)),
                "Rule": rules,
                "Credit Impact": "/".join(sorted(credit_impacts)),
                "Equity Impact": "/".join(sorted(equity_impacts)),
                "Filter Source": "database",
            })

    return dates_to_filter, audit_rows

def detect_stale_returns(returns_df):
    """Layer 2 safety net: detect days where SOME (not all) assets have ~0% returns.
    A 0% return from pct_change() means Bloomberg carried forward a stale price.
    This catches any holidays NOT in the database."""
    stale_mask = returns_df.abs() < 1e-10
    n_stale_per_day = stale_mask.sum(axis=1)
    n_total = len(returns_df.columns)

    # Only flag days where SOME (but not ALL) assets are stale → mismatch
    # If ALL are stale, there's no mismatch (PCA sees consistent zero returns)
    mismatch_mask = (n_stale_per_day > 0) & (n_stale_per_day < n_total)
    mismatch_dates = returns_df.index[mismatch_mask]

    audit_rows = []
    for date in mismatch_dates:
        stale_assets = list(returns_df.columns[stale_mask.loc[date]])
        audit_rows.append({
            "Date": date,
            "Day of Week": date.strftime("%A"),
            "Region": "Unknown",
            "Rule": f"Stale: {', '.join(stale_assets)}",
            "Credit Impact": "unknown",
            "Equity Impact": "unknown",
            "Filter Source": "stale-detection",
        })

    return list(mismatch_dates), audit_rows

def filter_calendar_misalignment(returns_df, selected_assets, start_year, end_year):
    """Orchestrate calendar filtering: database holidays + stale-return safety net.

    Returns: (clean_returns, audit_df, n_database_filtered, n_stale_filtered)
    """
    # 1. Merge manual (ad-hoc) + auto-generated (recurring) holidays
    all_holidays = MARKET_HOLIDAYS + generate_recurring_holidays(start_year, end_year)

    # 2. Layer 1: Get dates to filter from holiday database
    db_dates, db_audit = get_dates_to_filter(all_holidays, selected_assets)
    db_timestamps = pd.DatetimeIndex(db_dates)
    dates_in_data = returns_df.index.intersection(db_timestamps)
    clean_returns = returns_df.drop(dates_in_data, errors="ignore")
    n_db = len(dates_in_data)

    # 3. Layer 2: Detect remaining stale returns the database missed
    stale_dates, stale_audit = detect_stale_returns(clean_returns)
    clean_returns = clean_returns.drop(stale_dates, errors="ignore")
    n_stale = len(stale_dates)

    # 4. Build combined audit log
    all_audit = db_audit + stale_audit
    if all_audit:
        audit_df = pd.DataFrame(all_audit).sort_values("Date").reset_index(drop=True)
    else:
        audit_df = pd.DataFrame()

    return clean_returns, audit_df, n_db, n_stale

def check_roll_date_reminder():
    today = datetime.now()
    if (today.month == 3 and 15 <= today.day <= 31) or (today.month == 9 and 15 <= today.day <= 30):
        current_year = today.year
        cdx_dates = ROLL_DATES["CDX IG"]
        latest_year = int(cdx_dates[-1][:4]) if cdx_dates else 0
        if current_year > latest_year or (current_year == latest_year and today.month == 9 and "09" not in cdx_dates[-1]):
            return True
    return False

def load_data_bql(assets, field_map, start_date, end_date):
    all_dfs = []
    for asset in assets:
        ticker = ASSET_TICKERS.get(asset, asset)
        data_item = bq.data.px_last(dates=bq.func.range(start_date, end_date))
        request = bql.Request(ticker, {"VALUE": data_item})
        response = bq.execute(request)
        df = response.single().df()
        df = df.reset_index()
        df = df[["DATE", "VALUE"]].rename(columns={"VALUE": asset})
        df = df.set_index("DATE")
        all_dfs.append(df)
    result = pd.concat(all_dfs, axis=1)
    return result.sort_index().dropna()

def calculate_kmo(returns):
    corr = returns.corr().values
    try:
        inv_corr = np.linalg.inv(corr)
    except:
        inv_corr = np.linalg.inv(corr + 0.001 * np.eye(len(corr)))
    diag = np.diag(inv_corr)
    partial = -inv_corr / np.sqrt(np.outer(diag, diag))
    np.fill_diagonal(partial, 1.0)
    r_sq = corr ** 2
    p_sq = partial ** 2
    np.fill_diagonal(r_sq, 0)
    np.fill_diagonal(p_sq, 0)
    return r_sq.sum() / (r_sq.sum() + p_sq.sum())

def suggest_cohorts(returns, min_size=4, max_size=8, top_n=5):
    all_assets = list(returns.columns)
    results = []
    for size in range(min_size, min(max_size+1, len(all_assets)+1)):
        for combo in combinations(all_assets, size):
            assets = list(combo)
            n_credit = sum(1 for a in assets if a in CREDIT_ASSETS)
            n_equity = len(assets) - n_credit
            if n_credit >= 2 and n_equity >= 2:
                kmo = calculate_kmo(returns[assets])
                results.append({"assets": assets, "size": size, "kmo": kmo, "credit": n_credit, "equity": n_equity})
    results.sort(key=lambda x: x["kmo"], reverse=True)
    return results[:top_n]

def run_validation_tests(returns):
    results = {}
    kmo = calculate_kmo(returns)
    results["KMO"] = {"value": kmo, "status": "PASS" if kmo >= 0.8 else "WARN" if kmo >= 0.6 else "FAIL"}
    corr = returns.corr()
    mask = np.triu(np.ones_like(corr, dtype=bool), k=1)
    avg_corr = corr.where(mask).stack().abs().mean()
    results["Correlation"] = {"value": avg_corr, "status": "PASS" if avg_corr >= 0.3 else "FAIL"}
    results["corr_matrix"] = corr
    window = 60
    assets = returns.columns.tolist()
    stds = []
    for i in range(len(assets)):
        for j in range(i+1, len(assets)):
            roll_corr = returns[assets[i]].rolling(window=window).corr(returns[assets[j]]).dropna()
            stds.append(roll_corr.std())
    avg_std = np.mean(stds)
    results["Stability"] = {"value": avg_std, "status": "PASS" if avg_std < 0.2 else "WARN" if avg_std < 0.4 else "FAIL"}
    lag_corrs = []
    for i in range(len(assets)):
        for j in range(len(assets)):
            if i != j:
                lag_corr = returns[assets[i]].shift(1).corr(returns[assets[j]])
                lag_corrs.append(abs(lag_corr))
    avg_lag = np.mean(lag_corrs)
    results["LeadLag"] = {"value": avg_lag, "status": "PASS" if avg_lag < 0.1 else "WARN"}
    dist = 1 - corr.abs()
    dist_values = dist.values
    dist_values = (dist_values + dist_values.T) / 2
    np.fill_diagonal(dist_values, 0)
    condensed = squareform(dist_values)
    linkage_matrix = linkage(condensed, method="ward")
    labels = fcluster(linkage_matrix, t=linkage_matrix[:, 2].max() * 0.5, criterion="distance")
    n_clusters = len(set(labels))
    results["Clustering"] = {"value": n_clusters, "status": "PASS" if 2 <= n_clusters <= 4 else "WARN"}
    return results

def run_pca_analysis(returns, num_pcs=1, zscore_window=60, winsor_cap=3.0):
    credit = [a for a in returns.columns if a in CREDIT_ASSETS]
    equity = [a for a in returns.columns if a in EQUITY_ASSETS]
    returns = returns.replace([np.inf, -np.inf], np.nan).dropna()
    scaler = StandardScaler()
    data_scaled = scaler.fit_transform(returns)
    pca = PCA()
    pca_scores = pca.fit_transform(data_scaled)
    loadings = pd.DataFrame(pca.components_.T, columns=[f"PC{i+1}" for i in range(len(returns.columns))], index=returns.columns)
    var_explained = pd.Series(pca.explained_variance_ratio_, index=[f"PC{i+1}" for i in range(len(returns.columns))])
    expected = np.zeros_like(data_scaled)
    for i in range(num_pcs):
        beta = loadings[f"PC{i+1}"].values
        expected += np.outer(pca_scores[:, i], beta)
    residuals = data_scaled - expected

    # Winsorize extreme daily residuals before cumsumming
    # Without this, a single extreme day (FOMC, surprise event) injects an outsized
    # value into cumsum that permanently distorts the cumulative residual level.
    # winsor_cap=0 disables winsorization (no clipping)
    if winsor_cap > 0:
        residual_std = np.std(residuals, axis=0, keepdims=True)
        residual_std = np.where(residual_std < 1e-10, 1.0, residual_std)  # safety floor
        residual_clip = winsor_cap * residual_std
        residuals = np.clip(residuals, -residual_clip, residual_clip)

    # Cumulate daily residuals into mispricing levels
    # Each day's residual = how much the asset's return deviated from PC expectation
    # Summing over time: persistent deviations compound, random noise cancels out
    cumulative_residuals = pd.DataFrame(
        np.cumsum(residuals, axis=0),
        index=returns.index,
        columns=returns.columns
    )

    # Exponentially weighted z-score of cumulative residuals
    # EWM avoids the "cliff effect" of rolling windows: old observations fade gradually
    # instead of dropping off suddenly, so z-scores don't jump overnight when nothing moved
    # span parameter gives similar effective lookback to a rolling window but smooth
    ewm_mean = cumulative_residuals.ewm(span=zscore_window).mean()
    ewm_std = cumulative_residuals.ewm(span=zscore_window).std()
    z_scores = (cumulative_residuals - ewm_mean) / ewm_std
    z_scores = z_scores.dropna()

    # FLIP credit z-scores so that POSITIVE = RICH for ALL assets
    # Without flip: positive residual for credit = widened more than expected = CHEAP
    # After flip: positive z-score for credit = RICH (spreads too tight)
    # This makes interpretation consistent: positive Z = rich, negative Z = cheap
    for asset in credit:
        if asset in z_scores.columns:
            z_scores[asset] = -z_scores[asset]
    return {"z_scores": z_scores, "loadings": loadings, "variance": var_explained, "credit": credit, "equity": equity}

def get_ranking(z_scores, credit, as_of_date=None):
    if as_of_date is None:
        as_of_date = z_scores.index[-1]
    z_row = z_scores.loc[as_of_date]
    ranking = []
    for asset in z_scores.columns:
        z = z_row[asset]
        if abs(z) >= 2:
            signal = "VERY RICH" if z > 0 else "VERY CHEAP"
        elif abs(z) >= 1:
            signal = "Rich" if z > 0 else "Cheap"
        else:
            signal = "Fair"
        ranking.append({"Asset": asset, "Type": "Credit" if asset in credit else "Equity", "Z-Score": z, "Signal": signal})
    df = pd.DataFrame(ranking).sort_values("Z-Score", ascending=False)
    df["Rank"] = range(1, len(df) + 1)
    return df[["Rank", "Asset", "Type", "Z-Score", "Signal"]]

def format_trade_action(asset, position, asset_type):
    """
    Format trade action with proper buy/sell protection terminology.

    Z-SCORE CONVENTION (after credit flip in run_pca_analysis):
      - POSITIVE Z = RICH = asset outperformed vs model = expect mean-reversion DOWN
      - NEGATIVE Z = CHEAP = asset underperformed vs model = expect mean-reversion UP

    For Credit (spreads in bps):
      - RICH (positive Z) = spreads are TIGHTER than model expects
        → Expect spreads to WIDEN back to fair value
        → BUY protection to profit from widening
      - CHEAP (negative Z) = spreads are WIDER than model expects
        → Expect spreads to TIGHTEN back to fair value
        → SELL protection to profit from tightening

    For Equity (prices):
      - RICH (positive Z) = price is HIGHER than model expects
        → Expect price to FALL back to fair value
        → SELL index to profit from falling
      - CHEAP (negative Z) = price is LOWER than model expects
        → Expect price to RISE back to fair value
        → BUY index to profit from rising

    Parameters:
      position: 'short' (the rich asset) or 'long' (the cheap asset)
      asset_type: 'Credit' or 'Equity'
    """
    if asset_type == "Credit":
        if position == "short":
            # Rich credit = tight spreads, expect widening, buy protection to profit
            return f"Buy Protection on {asset}"
        else:
            # Cheap credit = wide spreads, expect tightening, sell protection to profit
            return f"Sell Protection on {asset}"
    else:  # Equity
        if position == "short":
            return f"Sell {asset}"
        else:
            return f"Buy {asset}"

def get_rv_opportunities(ranking):
    opps = {}
    for asset_type in ["Credit", "Equity"]:
        subset = ranking[ranking["Type"] == asset_type].sort_values("Z-Score", ascending=False)
        if len(subset) >= 2:
            rich = subset.iloc[0]
            cheap = subset.iloc[-1]
            opps[f"{asset_type} RV"] = {
                "Short": rich["Asset"],
                "Short Z": rich["Z-Score"],
                "Long": cheap["Asset"],
                "Long Z": cheap["Z-Score"],
                "Spread": rich["Z-Score"] - cheap["Z-Score"],
                "Short Action": format_trade_action(rich["Asset"], "short", asset_type),
                "Long Action": format_trade_action(cheap["Asset"], "long", asset_type),
            }
    return opps

def create_zscore_scatter(z_scores, asset1, asset2, lookback=756):
    """Z-Score scatter chart as HTML/SVG with dark theme and interactivity."""
    import json
    recent = z_scores.iloc[-lookback:]
    x_data = recent[asset1].values
    y_data = recent[asset2].values
    dates = recent.index
    n_pts = len(x_data)

    # Chart dimensions
    W, H = 700, 380
    M = {"top": 30, "right": 40, "bottom": 50, "left": 55}
    PW, PH = W - M["left"] - M["right"], H - M["top"] - M["bottom"]

    # Dynamic axis range — fits all data points with padding
    data_extent = max(np.abs(x_data).max(), np.abs(y_data).max(), 2.5)
    axis_limit = np.ceil(data_extent + 0.5)
    X_MIN, X_MAX = -axis_limit, axis_limit
    Y_MIN, Y_MAX = -axis_limit, axis_limit
    def sx(v): return M["left"] + (v - X_MIN) / (X_MAX - X_MIN) * PW
    def sy(v): return M["top"] + (1 - (v - Y_MIN) / (Y_MAX - Y_MIN)) * PH

    # Current values
    curr_x, curr_y = x_data[-1], y_data[-1]
    curr_date = dates[-1].strftime('%d%b%y')

    # Generate unique chart ID
    chart_id = _generate_chart_id()

    # Build data JSON for interactivity
    data_list = [{"x": float(x_data[i]), "y": float(y_data[i]), "date": dates[i].strftime('%d%b%y')} for i in range(n_pts)]
    data_json = json.dumps(data_list)

    html = _svg_chart_base(chart_id, W, H, M, f"{asset1} vs {asset2} Z-Score",
        f"Current: {asset1}={curr_x:.2f}, {asset2}={curr_y:.2f} ({curr_date})")

    # Grid lines — integer z-score levels within the dynamic range
    for v in range(int(-axis_limit) + 1, int(axis_limit)):
        if v == 0:
            continue
        html += f'<line x1="{sx(X_MIN)}" x2="{sx(X_MAX)}" y1="{sy(v)}" y2="{sy(v)}" stroke="{COLORS["gridLines"]}" stroke-width="0.5"/>'
        html += f'<line x1="{sx(v)}" x2="{sx(v)}" y1="{sy(Y_MIN)}" y2="{sy(Y_MAX)}" stroke="{COLORS["gridLines"]}" stroke-width="0.5"/>'

    # Reference lines at ±1, ±2, ±3, ±4, ±5 (only if within range)
    ref_styles = {
        1: (COLORS["textDim"], "0.8", "4,4"),
        2: (COLORS["textMuted"], "1", "2,2"),
        3: (COLORS["orange"], "0.8", "3,3"),
        4: (COLORS["red"], "0.8", "2,4"),
        5: (COLORS["red"], "1", "1,2"),
    }
    for level, (color, width, dash) in ref_styles.items():
        for v in [level, -level]:
            if Y_MIN <= v <= Y_MAX:
                html += f'<line x1="{sx(X_MIN)}" x2="{sx(X_MAX)}" y1="{sy(v)}" y2="{sy(v)}" stroke="{color}" stroke-width="{width}" stroke-dasharray="{dash}"/>'
                html += f'<line x1="{sx(v)}" x2="{sx(v)}" y1="{sy(Y_MIN)}" y2="{sy(Y_MAX)}" stroke="{color}" stroke-width="{width}" stroke-dasharray="{dash}"/>'

    # Axes
    html += f'<line x1="{M["left"]}" x2="{M["left"]+PW}" y1="{M["top"]+PH}" y2="{M["top"]+PH}" stroke="#444"/>'
    html += f'<line x1="{M["left"]}" x2="{M["left"]}" y1="{M["top"]}" y2="{M["top"]+PH}" stroke="#444"/>'

    # Zero lines
    html += f'<line x1="{sx(0)}" x2="{sx(0)}" y1="{sy(Y_MIN)}" y2="{sy(Y_MAX)}" stroke="{COLORS["textMuted"]}" stroke-width="1"/>'
    html += f'<line x1="{sx(X_MIN)}" x2="{sx(X_MAX)}" y1="{sy(0)}" y2="{sy(0)}" stroke="{COLORS["textMuted"]}" stroke-width="1"/>'

    # Scatter points (time gradient: older = dimmer)
    html += f'<g clip-path="url(#{chart_id}_clip)">'
    for i in range(n_pts):

    # X-axis ticks and labels (dynamic range)
    for v in range(int(-axis_limit), int(axis_limit) + 1):
        html += f'<line x1="{sx(v)}" x2="{sx(v)}" y1="{M["top"]+PH}" y2="{M["top"]+PH+5}" stroke="#444"/>'

    html += _svg_interactive_layer(chart_id, W, H, M, X_MIN, X_MAX, Y_MIN, Y_MAX, data_json, asset1, asset2)

    return html

def create_zscore_spread_chart(z_scores, asset1, asset2, lookback=756, ma_windows=None):
    """Z-Score spread time series as HTML/SVG with dark theme and interactivity."""
    import json
    if ma_windows is None:
        ma_windows = []  # Default: raw signal only (cumulative residuals are already smooth)

    recent = z_scores.iloc[-lookback:]
    spread = recent[asset1] - recent[asset2]
    dates = recent.index.tolist()
    y_data = spread.values
    n_pts = len(y_data)

    # Calculate smoothed moving averages for each selected window
    spread_series = pd.Series(y_data)
    smoothed_lines = {}
    for window in ma_windows:
        smoothed_lines[window] = spread_series.rolling(window=window, min_periods=1).mean().values

    # Chart dimensions
    W, H = 700, 300
    M = {"top": 30, "right": 40, "bottom": 50, "left": 55}
    PW, PH = W - M["left"] - M["right"], H - M["top"] - M["bottom"]

    # Scale functions — dynamic range, fits all data
    y_min = np.floor(y_data.min()) - 0.5
    y_max = np.ceil(y_data.max()) + 0.5
    def sx(i): return M["left"] + i / (n_pts - 1) * PW
    def sy(v): return M["top"] + (1 - (v - y_min) / (y_max - y_min)) * PH

    # Stats
    hi_idx, lo_idx = np.argmax(y_data), np.argmin(y_data)
    curr_val = y_data[-1]
    curr_date = dates[-1].strftime('%d%b%y')

    signal_color = COLORS["red"] if curr_val > 0 else COLORS["green"]

def load_regime_data(x_asset, y_asset, start_date, end_date):
    """
    Load data for any two assets for regime map.
    For equity: calculate % deviation from 200d MA
    For credit: use raw spread value
    Returns df with columns: X_VALUE, Y_VALUE
    """
    # Load X-axis asset
    x_ticker = ASSET_TICKERS.get(x_asset, x_asset)
    x_request = bql.Request(x_ticker, {"PX": bq.data.px_last(dates=bq.func.range(start_date, end_date))})
    x_response = bq.execute(x_request)
    x_df = x_response.single().df().reset_index()
    x_df = x_df[["DATE", "PX"]].rename(columns={"PX": "X_RAW"}).set_index("DATE")

    # Load Y-axis asset
    y_ticker = ASSET_TICKERS.get(y_asset, y_asset)
    y_request = bql.Request(y_ticker, {"PX": bq.data.px_last(dates=bq.func.range(start_date, end_date))})
    y_response = bq.execute(y_request)
    y_df = y_response.single().df().reset_index()
    y_df = y_df[["DATE", "PX"]].rename(columns={"PX": "Y_RAW"}).set_index("DATE")

    df = pd.concat([x_df, y_df], axis=1).dropna()

    # Process X-axis value
    if x_asset in CREDIT_ASSETS:
        # Credit: use raw spread value (bps)
        df["X_VALUE"] = df["X_RAW"]
    else:
        # Equity: calculate % deviation from 200d MA
        df["X_MA200"] = df["X_RAW"].rolling(window=200).mean()
        df["X_VALUE"] = ((df["X_RAW"] / df["X_MA200"]) - 1) * 100

    # Process Y-axis value
    if y_asset in CREDIT_ASSETS:
        # Credit: use raw spread value (bps)
        df["Y_VALUE"] = df["Y_RAW"]
    else:
        # Equity: calculate % deviation from 200d MA
        df["Y_MA200"] = df["Y_RAW"].rolling(window=200).mean()
        df["Y_VALUE"] = ((df["Y_RAW"] / df["Y_MA200"]) - 1) * 100

    df = df.dropna()
    return df

def regime_ols(x, y):
    n = len(x)
    sx, sy, sxy, sxx = np.sum(x), np.sum(y), np.sum(x * y), np.sum(x * x)
    b = (n * sxy - sx * sy) / (n * sxx - sx * sx)
    a = (sy - b * sx) / n
    return {"intercept": a, "slope": b}

def regime_kernel(x_data, y_data, x_grid, bandwidth=4):
    y_pred = []
    for xg in x_grid:
        u = (x_data - xg) / bandwidth
        w = np.exp(-0.5 * u * u)
        w_sum = np.sum(w)
        y_pred.append(np.sum(w * y_data) / w_sum if w_sum > 0 else 0)
    return np.array(y_pred)

def sigmoid_val(x, params):
    floor, ceiling, k, midpoint = params
    arg = np.clip(k * (x - midpoint), -500, 500)
    return floor + (ceiling - floor) / (1 + np.exp(arg))

def sigmoid_deriv(x, params):
    floor, ceiling, k, midpoint = params
    arg = np.clip(k * (x - midpoint), -500, 500)
    exp_term = np.exp(arg)
    return -(ceiling - floor) * k * exp_term / (1 + exp_term) ** 2

def fit_sigmoid(x_data, y_data):
    """
    Fit sigmoid curve with data-driven constraints.
    Works for any asset pair (equity vs credit, credit vs credit, etc.)

    Key insight: the sign of k determines the sigmoid direction:
      k > 0 = DECREASING sigmoid (e.g. SPX deviation vs CDX IG - negative correlation)
      k < 0 = INCREASING sigmoid (e.g. CDX IG vs CDX HY - positive correlation)
    We detect the correlation direction and use the appropriate sign.
    """
    # Calculate data-driven bounds
    x_min, x_max = np.min(x_data), np.max(x_data)
    y_min, y_max = np.min(y_data), np.max(y_data)
    x_range = x_max - x_min
    y_range = y_max - y_min
    x_mean = np.mean(x_data)

    # Floor/ceiling bounds based on Y data range (with some padding)
    floor_min, floor_max = y_min - 0.1 * y_range, y_min + 0.3 * y_range
    ceil_min, ceil_max = y_max - 0.3 * y_range, y_max + 0.1 * y_range
    min_range = 0.3 * y_range  # Minimum floor-ceiling gap

    # Midpoint bounds based on X data range
    mid_min, mid_max = x_min - 0.2 * x_range, x_max + 0.2 * x_range

    # Steepness bounds - allow both positive AND negative k
    # Positive k = decreasing sigmoid (equity vs credit)
    # Negative k = increasing sigmoid (credit vs credit, or any positive correlation)
    k_scale = 1.0 / max(x_range, 1)  # Normalize by x range
    k_mag_min = 0.01 * k_scale   # Minimum absolute steepness
    k_mag_max = 5.0 * k_scale    # Maximum absolute steepness

    # Detect relationship direction to choose starting k sign
    correlation = np.corrcoef(x_data, y_data)[0, 1]

    def cost(params):
        floor, ceiling, k, midpoint = params
        # Check bounds
        if floor < floor_min or floor > floor_max:
            return 1e15
        if ceiling < ceil_min or ceiling > ceil_max:
            return 1e15
        if ceiling - floor < min_range:
            return 1e15
        if abs(k) < k_mag_min or abs(k) > k_mag_max:
            return 1e15
        if midpoint < mid_min or midpoint > mid_max:
            return 1e15
        return np.sum((sigmoid_val(x_data, params) - y_data) ** 2)

    # Generate data-driven starting points with correct k sign
    floor_start = y_min + 0.1 * y_range
    ceil_start = y_max - 0.1 * y_range
    mid_start = x_mean

    if correlation > 0:
        # Positive correlation (e.g. CDX IG vs CDX HY) → increasing sigmoid → negative k
        k_signs = [-0.3, -0.5, -1.0, -1.5, -2.0]
    else:
        # Negative correlation (e.g. SPX vs CDX IG) → decreasing sigmoid → positive k
        k_signs = [0.3, 0.5, 1.0, 1.5, 2.0]

    starts = [
        [floor_start, ceil_start, k_signs[1] * k_scale, mid_start],
        [floor_start * 0.95, ceil_start * 1.05, k_signs[0] * k_scale, mid_start - 0.1 * x_range],
        [floor_start * 1.05, ceil_start * 0.95, k_signs[2] * k_scale, mid_start + 0.1 * x_range],
        [floor_start, ceil_start, k_signs[3] * k_scale, mid_start],
        [floor_start * 0.9, ceil_start * 1.1, k_signs[4] * k_scale, mid_start],
    ]

    best_result, best_cost = None, 1e15
    for start in starts:
        try:
            result = minimize(cost, start, method='Nelder-Mead', options={'maxiter': 10000})
            if result.fun < best_cost:
                best_cost, best_result = result.fun, result.x
        except:
            continue

    if best_result is None:
        # Fallback
        best_result = [floor_start, ceil_start, k_signs[1] * k_scale, mid_start]

    return {"floor": best_result[0], "ceiling": best_result[1], "steepness": best_result[2], "midpoint": best_result[3]}

def kernel_deriv(x_data, y_data, x_point, bandwidth=4, delta=0.01):
    y_minus = regime_kernel(x_data, y_data, [x_point - delta], bandwidth)[0]
    y_plus = regime_kernel(x_data, y_data, [x_point + delta], bandwidth)[0]
    return (y_plus - y_minus) / (2 * delta)

def calc_beta_table(x_data, y_data, ols_result, sig_params, bandwidth=4, x_label="X", x_unit="%",
                    sigma_x_data=None, model_for_ymove="Sigmoid"):
    """
    Calculate beta table with customizable sigma calculation and Y_Move model.

    sigma_x_data: Optional subset of x_data to use for sigma calculation (e.g., stress periods)
    model_for_ymove: "Sigmoid", "Kernel", or "OLS" - which model to use for Y_Move column
    """
    # Use provided sigma data or default to full x_data
    sigma_data = sigma_x_data if sigma_x_data is not None else x_data
    x_mean, x_std = np.mean(sigma_data), np.std(sigma_data)

    sig_p = [sig_params["floor"], sig_params["ceiling"], sig_params["steepness"], sig_params["midpoint"]]

    # Calculate Y at mean X position (reference point) using selected model
    if model_for_ymove == "Sigmoid":
        y_at_mean = sigmoid_val(x_mean, sig_p)
    elif model_for_ymove == "Kernel":
        y_at_mean = regime_kernel(x_data, y_data, [x_mean], bandwidth)[0]
    else:  # OLS
        y_at_mean = ols_result["intercept"] + ols_result["slope"] * x_mean

    points = [
        ("-3σ", x_mean - 3*x_std), ("-2σ", x_mean - 2*x_std), ("-1σ", x_mean - x_std),
        ("Mean", x_mean),
        ("+1σ", x_mean + x_std), ("+2σ", x_mean + 2*x_std), ("+3σ", x_mean + 3*x_std),
    ]
    rows = []
    for label, xp in points:
        # Calculate expected Y at this X point using selected model
        if model_for_ymove == "Sigmoid":
            y_at_point = sigmoid_val(xp, sig_p)
        elif model_for_ymove == "Kernel":
            y_at_point = regime_kernel(x_data, y_data, [xp], bandwidth)[0]
        else:  # OLS
            y_at_point = ols_result["intercept"] + ols_result["slope"] * xp

        y_move = y_at_point - y_at_mean

        rows.append({
            "Shock": label,
            "X_Val": f"{xp:.1f}",
            "Y_Move": f"{y_move:+.1f}",
            "OLS": f"{ols_result['slope']:.2f}",
            "Kern": f"{kernel_deriv(x_data, y_data, xp, bandwidth):.2f}",
            "Sig": f"{sigmoid_deriv(xp, sig_p):.2f}",
        })
    return pd.DataFrame(rows), x_mean, x_std

        html += f'<line x1="{sx(v)}" x2="{sx(v)}" y1="{M["top"]}" y2="{M["top"]+PH}" stroke="{COLORS["gridLines"]}" stroke-width="0.5"/>'
    y_step = 20 if y_max - y_min > 100 else 10
    for v in np.arange(np.ceil(y_min/y_step)*y_step, y_max+1, y_step):
        html += f'<line x1="{M["left"]}" x2="{M["left"]+PW}" y1="{sy(v)}" y2="{sy(v)}" stroke="{COLORS["gridLines"]}" stroke-width="0.5"/>'

    # Axes
    html += f'<line x1="{M["left"]}" x2="{M["left"]+PW}" y1="{M["top"]+PH}" y2="{M["top"]+PH}" stroke="#444"/>'
    html += f'<line x1="{M["left"]}" x2="{M["left"]}" y1="{M["top"]}" y2="{M["top"]+PH}" stroke="#444"/>'

    # Zero line (vertical at x=0)
    if x_min <= 0 <= x_max:

    for i in range(n_pts):
        opacity = 0.3 + 0.5 * (i / n_pts)
        html += f'<circle cx="{sx(x_data[i]):.1f}" cy="{sy(y_data[i]):.1f}" r="2.2" fill="{COLORS["scatterFill"]}" fill-opacity="{opacity:.2f}" stroke="{COLORS["scatterStroke"]}" stroke-width="0.5"/>'

    if show_ols:
        ols_y_vals = ols_result["intercept"] + ols_result["slope"] * x_grid
        path_d = " ".join([f"{'M' if i==0 else 'L'}{sx(x_grid[i]):.1f},{sy(ols_y_vals[i]):.1f}" for i in range(len(x_grid))])
        html += f'<path d="{path_d}" fill="none" stroke="{COLORS["orange"]}" stroke-width="1.5" stroke-dasharray="5,3" clip-path="url(#{chart_id}_clip)" opacity="0.9"/>'

        path_d = " ".join([f"{'M' if i==0 else 'L'}{sx(x_grid[i]):.1f},{sy(kernel_y[i]):.1f}" for i in range(len(x_grid))])

        html += f'<path d="{path_d}" fill="none" stroke="{COLORS["purple"]}" stroke-width="2.5" clip-path="url(#{chart_id}_clip)" opacity="0.95"/>'

    cx, cy = sx(curr_x), sy(curr_y)
    html += f'<polygon points="{cx-7},{cy} {cx},{cy-7} {cx+7},{cy} {cx},{cy+7}" fill="{COLORS["gold"]}" stroke="#000" stroke-width="1"/>'
    html += f'<text x="{cx+12}" y="{cy+4}" font-size="10" fill="{COLORS["gold"]}">Current</text>'

    for v in np.arange(np.ceil(x_min/x_step)*x_step, x_max+1, x_step):

def update_selected_display():
    selected_list.options = state.selected_assets

def on_add(b):
    asset = available_list.value
    if asset and asset not in state.selected_assets:
        state.selected_assets.append(asset)
        state.field_map[asset] = "PX_LAST"
        update_selected_display()

def on_remove(b):
    if selected_list.value:
        asset = selected_list.value
        if asset in state.selected_assets:
            state.selected_assets.remove(asset)
            if asset in state.field_map:
                del state.field_map[asset]
            update_selected_display()

def on_load(b):
    with data_output:
        clear_output(wait=True)
        if len(state.selected_assets) < 3:
            print("Add at least 3 assets")
            return
        if check_roll_date_reminder():
            display(HTML("<div style='background:#fff3cd;padding:10px;border-radius:5px;margin-bottom:10px;'><b>⚠️ REMINDER:</b> Update ROLL_DATES list!</div>"))
        state.raw_data = load_data_bql(state.selected_assets, state.field_map, start_date.value.strftime("%Y-%m-%d"), end_date.value.strftime("%Y-%m-%d"))
        # Compute returns FIRST on full price data, THEN remove discontinuity dates
        # This isolates artificial returns to the exact roll date for surgical removal
        # Credit: absolute change (bps) — pct_change on a 50bp spread distorts signals
        # Equity: percentage change — standard for price indices
        credit_cols = [a for a in state.selected_assets if a in CREDIT_ASSETS]
        equity_cols = [a for a in state.selected_assets if a in EQUITY_ASSETS]
        state.returns = pd.DataFrame(index=state.raw_data.index)
        for col in credit_cols:
            state.returns[col] = state.raw_data[col].diff()
        for col in equity_cols:
            state.returns[col] = state.raw_data[col].pct_change()
        state.returns = state.returns.iloc[1:]
        state.returns = state.returns.replace([np.inf, -np.inf], np.nan).dropna()
        discontinuity_dates = get_all_discontinuity_dates()
        dates_to_remove = state.returns.index.intersection(discontinuity_dates)
        if len(dates_to_remove) > 0:
            state.returns = state.returns.drop(dates_to_remove)
            print(f"Removed {len(dates_to_remove)} roll/recon date returns")
        start_yr = start_date.value.year
        end_yr = end_date.value.year
        state.returns, audit, n_db, n_stale = filter_calendar_misalignment(
            state.returns, state.selected_assets, start_yr, end_yr
        )
        state.calendar_audit = audit
        print(f"Calendar filter: {n_db} known holidays + {n_stale} detected stale = {n_db + n_stale} days removed")
        if len(audit) > 0:
            for _, row in audit.head(5).iterrows():
                print(f"  {row['Date'].strftime('%Y-%m-%d')} ({row['Day of Week'][:3]}) - {row['Rule']} [{row['Filter Source']}]")
        print(f"Loaded {len(state.returns)} days for {len(state.selected_assets)} assets")
        # Update Z-Score pair dropdowns
        pair_asset1.options = state.selected_assets
        pair_asset2.options = state.selected_assets
        if len(state.selected_assets) >= 2:
            pair_asset1.value = state.selected_assets[0]
            pair_asset2.value = state.selected_assets[1]
        # Update Regime Map dropdowns
        regime_x_asset.options = state.selected_assets
        regime_y_asset.options = state.selected_assets
        if len(state.selected_assets) >= 2:
            # Default to first equity and first credit if available
            equities = [a for a in state.selected_assets if a in EQUITY_ASSETS]
            credits = [a for a in state.selected_assets if a in CREDIT_ASSETS]
            regime_x_asset.value = equities[0] if equities else state.selected_assets[0]
            regime_y_asset.value = credits[0] if credits else state.selected_assets[1]

def on_run_tests(b):
    with tests_output:
        clear_output(wait=True)
        if state.returns is None:
            print("Load data first")
            return
        state.test_results = run_validation_tests(state.returns)
        for test, result in state.test_results.items():
            if test == "corr_matrix":
                continue
            color = "#2a9d8f" if result["status"] == "PASS" else "#e63946" if result["status"] == "FAIL" else "#f4a261"
            display(HTML(f"<span style='color:{color};font-weight:bold;'>{test}: {result['value']:.3f} [{result['status']}]</span>"))

def on_suggest(b):
    with cohort_output:
        clear_output(wait=True)
        if state.returns is None:
            print("Load data first")
            return
        print("Finding optimal cohorts...")
        state.suggested_cohorts = suggest_cohorts(state.returns)
        for i, c in enumerate(state.suggested_cohorts):
            display(HTML(f"<b>Option {i+1}:</b> KMO={c['kmo']:.3f} | {c['credit']}C/{c['equity']}E | {', '.join(c['assets'])}"))

def on_run_pca(b):
    with pca_output:
        clear_output(wait=True)
        if state.returns is None:
            print("Load data first")
            return
        state.pca_results = run_pca_analysis(state.returns, num_pcs=num_pcs.value, zscore_window=zscore_window.value, winsor_cap=winsor_cap_slider.value)
        var = state.pca_results["variance"]
        print(f"PC1: {var.iloc[0]:.1%} | PC2: {var.iloc[1]:.1%} | PC3: {var.iloc[2]:.1%}")
    with results_output:
        clear_output(wait=True)
        z = state.pca_results["z_scores"]
        ranking = get_ranking(z, state.pca_results["credit"])
        display(ranking)
        opps = get_rv_opportunities(ranking)
        for trade, d in opps.items():
            display(HTML(f"<b>{trade}:</b> {d['Short Action']} ({d['Short Z']:+.2f}) vs {d['Long Action']} ({d['Long Z']:+.2f}) | Spread: {d['Spread']:.2f}"))

def on_zscore_update(b):
    # Clear all z-score outputs
    for out in [zscore_output, legs_chart_output, zscore_spread_output]:
        with out:
            clear_output(wait=True)

    if state.pca_results is None:
        with zscore_output:
            print("Run PCA first")
        return
    a1, a2 = pair_asset1.value, pair_asset2.value
    if not a1 or not a2:
        with zscore_output:
            print("Select both assets")
        return
    z = state.pca_results["z_scores"]

    # Update legs dropdown options for selected pair
    legs_dropdown.options = ["Both", a1, a2]
    legs_dropdown.value = "Both"

    # Scatter chart
    with zscore_output:
        display(HTML(create_zscore_scatter(z, a1, a2)))

    # Individual legs chart
    update_legs_chart()

    # Spread chart
    ma_windows = []
    if zscore_ma_5d.value:
        ma_windows.append(5)
    if zscore_ma_10d.value:
        ma_windows.append(10)
    if zscore_ma_20d.value:
        ma_windows.append(20)
    if not ma_windows:
        ma_windows = [20]
    with zscore_spread_output:
        display(HTML(create_zscore_spread_chart(z, a1, a2, ma_windows=ma_windows)))

def on_regime_load(b):
    with regime_load_output:
        clear_output(wait=True)
        print("Loading...")
    try:
        # Store selected assets in state
        state.regime_x_asset = regime_x_asset.value
        state.regime_y_asset = regime_y_asset.value

        state.regime_df = load_regime_data(
            state.regime_x_asset,
            state.regime_y_asset,
            regime_start.value.strftime("%Y-%m-%d"),
            regime_end.value.strftime("%Y-%m-%d")
        )
        x_data, y_data = state.regime_df["X_VALUE"].values, state.regime_df["Y_VALUE"].values
        state.regime_ols = regime_ols(x_data, y_data)
        state.regime_sigmoid = fit_sigmoid(x_data, y_data)

        # Dynamically update slider bounds based on data
        y_min, y_max = np.min(y_data), np.max(y_data)
        y_range = max(y_max - y_min, 1)  # Avoid zero range
        x_min, x_max = np.min(x_data), np.max(x_data)
        x_range = max(x_max - x_min, 1)  # Avoid zero range

        # Calculate new bounds
        floor_min_new = y_min - 0.2 * y_range
        floor_max_new = y_min + 0.5 * y_range
        ceil_min_new = y_max - 0.5 * y_range
        ceil_max_new = y_max + 0.2 * y_range
        k_scale = 1.0 / max(x_range, 1)
        steep_min_new = -5.0 * k_scale
        steep_max_new = 5.0 * k_scale
        mid_min_new = x_min - 0.3 * x_range
        mid_max_new = x_max + 0.3 * x_range

        # Reset sliders to wide range first to avoid min > max errors
        # Then set to actual bounds
        regime_floor.min = min(floor_min_new, regime_floor.min, 0)
        regime_floor.max = max(floor_max_new, regime_floor.max, 1000)
        regime_floor.min = floor_min_new
        regime_floor.max = floor_max_new
        regime_floor.step = max(y_range / 100, 0.1)

        regime_ceiling.min = min(ceil_min_new, regime_ceiling.min, 0)
        regime_ceiling.max = max(ceil_max_new, regime_ceiling.max, 1000)
        regime_ceiling.min = ceil_min_new
        regime_ceiling.max = ceil_max_new
        regime_ceiling.step = max(y_range / 100, 0.1)

        regime_steep.min = min(steep_min_new, regime_steep.min)
        regime_steep.max = max(steep_max_new, regime_steep.max)
        regime_steep.min = steep_min_new
        regime_steep.max = steep_max_new
        regime_steep.step = max(0.1 * k_scale, 0.001)

        regime_mid.min = min(mid_min_new, regime_mid.min, -1000)
        regime_mid.max = max(mid_max_new, regime_mid.max, 1000)
        regime_mid.min = mid_min_new
        regime_mid.max = mid_max_new
        regime_mid.step = max(x_range / 100, 0.1)

        # Now set values (clamped to bounds)
        regime_floor.value = np.clip(state.regime_sigmoid["floor"], floor_min_new, floor_max_new)
        regime_ceiling.value = np.clip(state.regime_sigmoid["ceiling"], ceil_min_new, ceil_max_new)
        regime_steep.value = np.clip(state.regime_sigmoid["steepness"], steep_min_new, steep_max_new)
        regime_mid.value = np.clip(state.regime_sigmoid["midpoint"], mid_min_new, mid_max_new)

        # Create x_grid with appropriate padding
        x_padding = x_range * 0.1
        state.regime_x_grid = np.linspace(x_min - x_padding, x_max + x_padding, 300)
        state.regime_kernel_y = regime_kernel(x_data, y_data, state.regime_x_grid, regime_bandwidth.value)
        sig_p = [state.regime_sigmoid["floor"], state.regime_sigmoid["ceiling"], state.regime_sigmoid["steepness"], state.regime_sigmoid["midpoint"]]
        state.regime_sigmoid_y = sigmoid_val(state.regime_x_grid, sig_p)
        with regime_load_output:
            clear_output(wait=True)
            print(f"{len(state.regime_df)} days | floor={state.regime_sigmoid['floor']:.1f} ceil={state.regime_sigmoid['ceiling']:.1f}")
        update_regime_display()
    except Exception as e:
        with regime_load_output:
            clear_output(wait=True)
            print(f"Error: {e}")

def get_sigma_subset(df, lookback, stress_period):
    """
    Get subset of data for sigma calculation based on lookback period or stress period.
    Returns (subset_x_data, sigma_source_label)
    """
    x_data = df["X_VALUE"].values

    # If stress period is selected, use that
    if stress_period != "None" and stress_period in STRESS_PERIODS:
        start_str, end_str = STRESS_PERIODS[stress_period]
        start_dt = pd.to_datetime(start_str)
        end_dt = pd.to_datetime(end_str)
        mask = (df.index >= start_dt) & (df.index <= end_dt)
        if mask.sum() > 10:  # Need enough data points
            return df.loc[mask, "X_VALUE"].values, stress_period
        # Fall back to full if not enough data in stress period

    # Otherwise use lookback period
    if lookback == "Full History":
        return x_data, "Full History"
    elif lookback == "5Y":
        return x_data[-252*5:] if len(x_data) > 252*5 else x_data, "5Y"
    elif lookback == "3Y":
        return x_data[-252*3:] if len(x_data) > 252*3 else x_data, "3Y"
    elif lookback == "1Y":
        return x_data[-252:] if len(x_data) > 252 else x_data, "1Y"
    elif lookback == "6M":
        return x_data[-126:] if len(x_data) > 126 else x_data, "6M"
    else:
        return x_data, "Full History"

def update_regime_display():
    if state.regime_df is None:
        return
    x_data, y_data = state.regime_df["X_VALUE"].values, state.regime_df["Y_VALUE"].values
    state.regime_kernel_y = regime_kernel(x_data, y_data, state.regime_x_grid, regime_bandwidth.value)
    sig_p = [regime_floor.value, regime_ceiling.value, regime_steep.value, regime_mid.value]
    state.regime_sigmoid_y = sigmoid_val(state.regime_x_grid, sig_p)

    with regime_chart_output:
        clear_output(wait=True)
        chart_html = create_regime_chart(
            state.regime_df, state.regime_ols, state.regime_kernel_y, state.regime_sigmoid_y,
            state.regime_x_grid, regime_show_ols.value, regime_show_kernel.value, regime_show_sigmoid.value,
            x_asset=state.regime_x_asset, y_asset=state.regime_y_asset
        )
        display(HTML(chart_html))
    with regime_tables_output:
        clear_output(wait=True)
        sig_params = {"floor": regime_floor.value, "ceiling": regime_ceiling.value, "steepness": regime_steep.value, "midpoint": regime_mid.value}

        # Get sigma subset based on user selection
        sigma_x_data, sigma_source = get_sigma_subset(
            state.regime_df,
            regime_sigma_lookback.value,
            regime_stress_period.value
        )

        # Calculate beta table with customization
        beta_df, sigma_mean, sigma_std = calc_beta_table(
            x_data, y_data, state.regime_ols, sig_params, regime_bandwidth.value,
            sigma_x_data=sigma_x_data,
            model_for_ymove=regime_beta_model.value
        )

        stats_html = create_regime_stats_html(state.regime_df, x_asset=state.regime_x_asset, y_asset=state.regime_y_asset)
        beta_html = create_beta_table_html(
            beta_df, x_asset=state.regime_x_asset, y_asset=state.regime_y_asset,
            sigma_source=sigma_source, sigma_mean=sigma_mean, sigma_std=sigma_std,
            model_name=regime_beta_model.value
        )
        display(HTML(stats_html + beta_html))
