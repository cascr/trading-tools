def check_roll_date_reminder():
    today = datetime.now()
    if (today.month == 3 and 15 <= today.day <= 31) or (today.month == 9 and 15 <= today.day <= 30):
        current_year = today.year
        cdx_dates = ROLL_DATES["CDX IG"]
        latest_year = int(cdx_dates[-1][:4]) if cdx_dates else 0
        if current_year > latest_year or (current_year == latest_year and today.month == 9 and "09" not in cdx_dates[-1]):
            return True
    return False

def calculate_kmo(returns):
    corr = returns.corr().values
    try:
        inv_corr = np.linalg.inv(corr)
    except:
        inv_corr = np.linalg.inv(corr + 0.001 * np.eye(len(corr)))
    diag = np.diag(inv_corr)
    partial = -inv_corr / np.sqrt(np.outer(diag, diag))
    np.fill_diagonal(partial, 1.0)
    r_sq = corr ** 2
    p_sq = partial ** 2
    np.fill_diagonal(r_sq, 0)
    np.fill_diagonal(p_sq, 0)
    return r_sq.sum() / (r_sq.sum() + p_sq.sum())

def suggest_cohorts(returns, min_size=4, max_size=8, top_n=5):
    all_assets = list(returns.columns)
    results = []
    for size in range(min_size, min(max_size+1, len(all_assets)+1)):
        for combo in combinations(all_assets, size):
            assets = list(combo)
            n_credit = sum(1 for a in assets if a in CREDIT_ASSETS)
            n_equity = len(assets) - n_credit
            if n_credit >= 2 and n_equity >= 2:
                kmo = calculate_kmo(returns[assets])
                results.append({"assets": assets, "size": size, "kmo": kmo, "credit": n_credit, "equity": n_equity})
    results.sort(key=lambda x: x["kmo"], reverse=True)
    return results[:top_n]

def run_validation_tests(returns):
    results = {}
    kmo = calculate_kmo(returns)
    results["KMO"] = {"value": kmo, "status": "PASS" if kmo >= 0.8 else "WARN" if kmo >= 0.6 else "FAIL"}
    corr = returns.corr()
    mask = np.triu(np.ones_like(corr, dtype=bool), k=1)
    avg_corr = corr.where(mask).stack().abs().mean()
    results["Correlation"] = {"value": avg_corr, "status": "PASS" if avg_corr >= 0.3 else "FAIL"}
    results["corr_matrix"] = corr
    window = 60
    assets = returns.columns.tolist()
    stds = []
    for i in range(len(assets)):
        for j in range(i+1, len(assets)):
            roll_corr = returns[assets[i]].rolling(window=window).corr(returns[assets[j]]).dropna()
            stds.append(roll_corr.std())
    avg_std = np.mean(stds)
    results["Stability"] = {"value": avg_std, "status": "PASS" if avg_std < 0.2 else "WARN" if avg_std < 0.4 else "FAIL"}
    lag_corrs = []
    for i in range(len(assets)):
        for j in range(len(assets)):
            if i != j:
                lag_corr = returns[assets[i]].shift(1).corr(returns[assets[j]])
                lag_corrs.append(abs(lag_corr))
    avg_lag = np.mean(lag_corrs)
    results["LeadLag"] = {"value": avg_lag, "status": "PASS" if avg_lag < 0.1 else "WARN"}
    dist = 1 - corr.abs()
    dist_values = dist.values
    dist_values = (dist_values + dist_values.T) / 2
    np.fill_diagonal(dist_values, 0)
    condensed = squareform(dist_values)
    linkage_matrix = linkage(condensed, method="ward")
    labels = fcluster(linkage_matrix, t=linkage_matrix[:, 2].max() * 0.5, criterion="distance")
    n_clusters = len(set(labels))
    results["Clustering"] = {"value": n_clusters, "status": "PASS" if 2 <= n_clusters <= 4 else "WARN"}
    return results

def run_pca_analysis(returns, num_pcs=1):
    credit = [a for a in returns.columns if a in CREDIT_ASSETS]
    equity = [a for a in returns.columns if a in EQUITY_ASSETS]
    returns = returns.replace([np.inf, -np.inf], np.nan).dropna()
    scaler = StandardScaler()
    data_scaled = scaler.fit_transform(returns)
    pca = PCA()
    pca_scores = pca.fit_transform(data_scaled)
    loadings = pd.DataFrame(pca.components_.T, columns=[f"PC{i+1}" for i in range(len(returns.columns))], index=returns.columns)
    var_explained = pd.Series(pca.explained_variance_ratio_, index=[f"PC{i+1}" for i in range(len(returns.columns))])
    expected = np.zeros_like(data_scaled)
    for i in range(num_pcs):
        beta = loadings[f"PC{i+1}"].values
        expected += np.outer(pca_scores[:, i], beta)
    residuals = data_scaled - expected
    z_scores = pd.DataFrame((residuals - residuals.mean(axis=0)) / residuals.std(axis=0), index=returns.index, columns=returns.columns)
    for asset in credit:
        if asset in z_scores.columns:
            z_scores[asset] = -z_scores[asset]
    return {"z_scores": z_scores, "loadings": loadings, "variance": var_explained, "credit": credit, "equity": equity}

def get_ranking(z_scores, credit, as_of_date=None):
    if as_of_date is None:
        as_of_date = z_scores.index[-1]
    z_row = z_scores.loc[as_of_date]
    ranking = []
    for asset in z_scores.columns:
        z = z_row[asset]
        if abs(z) >= 2:
            signal = "VERY RICH" if z > 0 else "VERY CHEAP"
        elif abs(z) >= 1:
            signal = "Rich" if z > 0 else "Cheap"
        else:
            signal = "Fair"
        ranking.append({"Asset": asset, "Type": "Credit" if asset in credit else "Equity", "Z-Score": z, "Signal": signal})
    df = pd.DataFrame(ranking).sort_values("Z-Score", ascending=False)
    df["Rank"] = range(1, len(df) + 1)
    return df[["Rank", "Asset", "Type", "Z-Score", "Signal"]]

def get_rv_opportunities(ranking):
    opps = {}
    for asset_type in ["Credit", "Equity"]:
        subset = ranking[ranking["Type"] == asset_type].sort_values("Z-Score", ascending=False)
        if len(subset) >= 2:
            rich = subset.iloc[0]
            cheap = subset.iloc[-1]
            opps[f"{asset_type} RV"] = {"Short": rich["Asset"], "Short Z": rich["Z-Score"], "Long": cheap["Asset"], "Long Z": cheap["Z-Score"], "Spread": rich["Z-Score"] - cheap["Z-Score"]}
    return opps

def regime_ols(x, y):
    n = len(x)
    sx, sy, sxy, sxx = np.sum(x), np.sum(y), np.sum(x * y), np.sum(x * x)
    b = (n * sxy - sx * sy) / (n * sxx - sx * sx)
    a = (sy - b * sx) / n
    return {"intercept": a, "slope": b}

def regime_kernel(x_data, y_data, x_grid, bandwidth=4):
    y_pred = []
    for xg in x_grid:
        u = (x_data - xg) / bandwidth
        w = np.exp(-0.5 * u * u)
        w_sum = np.sum(w)
        y_pred.append(np.sum(w * y_data) / w_sum if w_sum > 0 else 0)
    return np.array(y_pred)

def sigmoid_val(x, params):
    floor, ceiling, k, midpoint = params
    arg = np.clip(k * (x - midpoint), -500, 500)
    return floor + (ceiling - floor) / (1 + np.exp(arg))

def sigmoid_deriv(x, params):
    floor, ceiling, k, midpoint = params
    arg = np.clip(k * (x - midpoint), -500, 500)
    exp_term = np.exp(arg)
    return -(ceiling - floor) * k * exp_term / (1 + exp_term) ** 2

def fit_sigmoid(x_data, y_data):
    def cost(params):
        floor, ceiling, k, midpoint = params
        if floor < 38 or floor > 60 or ceiling < 100 or ceiling > 300:
            return 1e15
        if k < 0.05 or k > 2.0 or midpoint < -25 or midpoint > 0:
            return 1e15
        if ceiling - floor < 60:
            return 1e15
        return np.sum((sigmoid_val(x_data, params) - y_data) ** 2)

    starts = [
        [48, 170, 0.15, -12], [47, 160, 0.20, -10], [50, 180, 0.25, -15],
        [46, 200, 0.10, -8], [48, 150, 0.30, -18], [45, 190, 0.12, -14],
    ]
    best_result, best_cost = None, 1e15
    for start in starts:
        result = minimize(cost, start, method='Nelder-Mead', options={'maxiter': 10000})
        if result.fun < best_cost:
            best_cost, best_result = result.fun, result.x
    return {"floor": best_result[0], "ceiling": best_result[1], "steepness": best_result[2], "midpoint": best_result[3]}

def kernel_deriv(x_data, y_data, x_point, bandwidth=4, delta=0.01):
    y_minus = regime_kernel(x_data, y_data, [x_point - delta], bandwidth)[0]
    y_plus = regime_kernel(x_data, y_data, [x_point + delta], bandwidth)[0]
    return (y_plus - y_minus) / (2 * delta)

def calc_beta_table(x_data, y_data, ols_result, sig_params, bandwidth=4):
    x_mean, x_std = np.mean(x_data), np.std(x_data)
    points = [
        ("-3σ", x_mean - 3*x_std), ("-2σ", x_mean - 2*x_std), ("-1σ", x_mean - x_std),
        ("Mean", x_mean),
        ("+1σ", x_mean + x_std), ("+2σ", x_mean + 2*x_std), ("+3σ", x_mean + 3*x_std),
    ]
    rows = []
    for label, xp in points:
        sig_p = [sig_params["floor"], sig_params["ceiling"], sig_params["steepness"], sig_params["midpoint"]]
        rows.append({
            "Pt": label, "SPX%": f"{xp:.1f}",
            "OLS": f"{ols_result['slope']:.2f}",
            "Kern": f"{kernel_deriv(x_data, y_data, xp, bandwidth):.2f}",
            "Sig": f"{sigmoid_deriv(xp, sig_p):.2f}",
        })
    return pd.DataFrame(rows)

    color_scale = bqp.ColorScale(colors=["#c0c0c0", "#1a1a2e"])
    date_labels = [d.strftime('%d%b%y') for d in dates]

    scatter = bqp.Scatter(x=x_data, y=y_data, scales={'x': x_scale, 'y': y_scale, 'color': color_scale},
                          color=colors, default_size=8, opacities=[0.6]*n_pts, names=date_labels)
    scatter.tooltip = bqp.Tooltip(fields=['names', 'x', 'y'], formats=['', '.1f', '.1f'],
                                   labels=['Date', 'SPX Dev%', 'CDX bps'])

    marks = [scatter]
    if show_ols:
        ols_y = ols_result["intercept"] + ols_result["slope"] * x_grid
        marks.append(bqp.Lines(x=x_grid, y=ols_y, scales={'x': x_scale, 'y': y_scale},
                               colors=[COLORS["ols"]], stroke_width=1.5, line_style="dashed"))
    if show_kernel:
        marks.append(bqp.Lines(x=x_grid, y=kernel_y, scales={'x': x_scale, 'y': y_scale},
                               colors=[COLORS["kernel"]], stroke_width=2))
    if show_sigmoid:
        marks.append(bqp.Lines(x=x_grid, y=sigmoid_y, scales={'x': x_scale, 'y': y_scale},
                               colors=[COLORS["sigmoid"]], stroke_width=2))
    # Zero line
    marks.append(bqp.Lines(x=[0, 0], y=[y_min, y_max], scales={'x': x_scale, 'y': y_scale},
                           colors=["#888"], stroke_width=1, line_style="dotted"))

    x_axis = bqp.Axis(scale=x_scale, label="SPX % Dev from 200d MA", grid_lines="none", tick_format=".0f")
    y_axis = bqp.Axis(scale=y_scale, orientation="vertical", label="CDX IG (bps)", grid_lines="solid", grid_color="#eee", tick_format=".0f")

    fig = bqp.Figure(marks=marks, axes=[x_axis, y_axis],
                     fig_margin={"top": 20, "bottom": 45, "left": 50, "right": 15},
                     background_style={"fill": "#fafafa"})
    fig.layout.width = "100%"
    fig.layout.height = "420px"
    return fig

def create_regime_stats_html(df):
    """Compact stats box for regime map"""
    y = df["CDX"].values
    dates = df.index
    hi, lo = np.argmax(y), np.argmin(y)
    return f"""<div style="background:#f8f9fa;border:1px solid #dee2e6;border-radius:3px;padding:6px 10px;font-family:monospace;font-size:9px;display:inline-block;">
    <table style="border-collapse:collapse;line-height:1.4;">
    <tr><td style="padding-right:8px;"><b>First</b> {dates[0].strftime('%d%b%y')}</td><td style="text-align:right;padding-right:12px;">{y[0]:.0f}</td>
        <td style="padding-right:8px;"><b>Last</b> {dates[-1].strftime('%d%b%y')}</td><td style="text-align:right;">{y[-1]:.0f}</td></tr>
    <tr><td style="padding-right:8px;"><b>High</b> {dates[hi].strftime('%d%b%y')}</td><td style="text-align:right;padding-right:12px;">{y[hi]:.0f}</td>
        <td style="padding-right:8px;"><b>Low</b> {dates[lo].strftime('%d%b%y')}</td><td style="text-align:right;">{y[lo]:.0f}</td></tr>
    <tr><td style="padding-right:8px;"><b>Mean</b></td><td style="text-align:right;padding-right:12px;">{np.mean(y):.0f}</td>
        <td style="padding-right:8px;"><b>Std</b></td><td style="text-align:right;">{np.std(y):.0f}</td></tr>
    </table></div>"""

def create_beta_table_html(beta_df):
    """Compact beta table HTML"""
    html = '<div style="background:#f8f9fa;border:1px solid #dee2e6;border-radius:3px;padding:6px 10px;font-family:monospace;font-size:9px;display:inline-block;margin-left:10px;">'
    html += '<table style="border-collapse:collapse;line-height:1.4;"><tr style="border-bottom:1px solid #ccc;">'
    html += '<th style="padding-right:8px;text-align:left;">Pt</th><th style="padding-right:8px;">SPX%</th>'
    html += '<th style="padding-right:8px;color:#f97316;">OLS</th><th style="padding-right:8px;color:#22d3ee;">Kern</th><th style="color:#a78bfa;">Sig</th></tr>'
    for _, row in beta_df.iterrows():
        html += f'<tr><td style="padding-right:8px;"><b>{row["Pt"]}</b></td><td style="padding-right:8px;text-align:right;">{row["SPX%"]}</td>'
        html += f'<td style="padding-right:8px;text-align:right;">{row["OLS"]}</td><td style="padding-right:8px;text-align:right;">{row["Kern"]}</td>'
        html += f'<td style="text-align:right;">{row["Sig"]}</td></tr>'
    html += '</table></div>'
    return html

def update_selected_display():
    selected_list.options = state.selected_assets

def on_add(b):
    asset = available_list.value
    if asset and asset not in state.selected_assets:
        state.selected_assets.append(asset)
        state.field_map[asset] = "PX_LAST"
        update_selected_display()

def on_remove(b):
    if selected_list.value:
        asset = selected_list.value
        if asset in state.selected_assets:
            state.selected_assets.remove(asset)
            if asset in state.field_map:
                del state.field_map[asset]
            update_selected_display()

def on_load(b):
    with data_output:
        clear_output(wait=True)
        if len(state.selected_assets) < 3:
            print("Add at least 3 assets")
            return
        if check_roll_date_reminder():
            display(HTML("<div style='background:#fff3cd;padding:10px;border-radius:5px;margin-bottom:10px;'><b>⚠️ REMINDER:</b> Update ROLL_DATES list!</div>"))
        state.raw_data = load_data_bql(state.selected_assets, state.field_map, start_date.value.strftime("%Y-%m-%d"), end_date.value.strftime("%Y-%m-%d"))
        roll_dates = get_all_roll_dates()
        dates_to_remove = state.raw_data.index.intersection(roll_dates)
        if len(dates_to_remove) > 0:
            state.raw_data = state.raw_data.drop(dates_to_remove)
            print(f"Removed {len(dates_to_remove)} roll dates")
        state.returns = state.raw_data.pct_change().dropna()
        state.returns = state.returns.replace([np.inf, -np.inf], np.nan).dropna()
        print(f"Loaded {len(state.returns)} days for {len(state.selected_assets)} assets")
        pair_asset1.options = state.selected_assets
        pair_asset2.options = state.selected_assets
        if len(state.selected_assets) >= 2:
            pair_asset1.value = state.selected_assets[0]
            pair_asset2.value = state.selected_assets[1]

def on_run_tests(b):
    with tests_output:
        clear_output(wait=True)
        if state.returns is None:
            print("Load data first")
            return
        state.test_results = run_validation_tests(state.returns)
        for test, result in state.test_results.items():
            if test == "corr_matrix":
                continue
            color = "#2a9d8f" if result["status"] == "PASS" else "#e63946" if result["status"] == "FAIL" else "#f4a261"
            display(HTML(f"<span style='color:{color};font-weight:bold;'>{test}: {result['value']:.3f} [{result['status']}]</span>"))

def on_suggest(b):
    with cohort_output:
        clear_output(wait=True)
        if state.returns is None:
            print("Load data first")
            return
        print("Finding optimal cohorts...")
        state.suggested_cohorts = suggest_cohorts(state.returns)
        for i, c in enumerate(state.suggested_cohorts):
            display(HTML(f"<b>Option {i+1}:</b> KMO={c['kmo']:.3f} | {c['credit']}C/{c['equity']}E | {', '.join(c['assets'])}"))

def on_run_pca(b):
    with pca_output:
        clear_output(wait=True)
        if state.returns is None:
            print("Load data first")
            return
        state.pca_results = run_pca_analysis(state.returns, num_pcs=num_pcs.value)
        var = state.pca_results["variance"]
        print(f"PC1: {var.iloc[0]:.1%} | PC2: {var.iloc[1]:.1%} | PC3: {var.iloc[2]:.1%}")
    with results_output:
        clear_output(wait=True)
        z = state.pca_results["z_scores"]
        ranking = get_ranking(z, state.pca_results["credit"])
        display(ranking)
        opps = get_rv_opportunities(ranking)
        for trade, d in opps.items():
            display(HTML(f"<b>{trade}:</b> Short {d['Short']} ({d['Short Z']:+.2f}) / Long {d['Long']} ({d['Long Z']:+.2f}) | Spread: {d['Spread']:.2f}"))

def on_zscore_update(b):
    with zscore_output:
        clear_output(wait=True)
        if state.pca_results is None:
            print("Run PCA first")
            return
        a1, a2 = pair_asset1.value, pair_asset2.value
        if not a1 or not a2:
            print("Select both assets")
            return
        z = state.pca_results["z_scores"]
        display(HTML(f"<b style='font-size:11px;'>{a1} vs {a2}</b>"))
        # Stacked charts
        scatter_fig = create_zscore_scatter(z, a1, a2)
        spread_fig = create_zscore_spread_chart(z, a1, a2)
        display(scatter_fig)
        display(spread_fig)
        # Stats below
        display(HTML(create_zscore_stats_html(z, a1, a2)))

def on_regime_load(b):
    with regime_load_output:
        clear_output(wait=True)
        print("Loading...")
    try:
        state.regime_df = load_regime_data(regime_start.value.strftime("%Y-%m-%d"), regime_end.value.strftime("%Y-%m-%d"))
        x_data, y_data = state.regime_df["SPX_DEV"].values, state.regime_df["CDX"].values
        state.regime_ols = regime_ols(x_data, y_data)
        state.regime_sigmoid = fit_sigmoid(x_data, y_data)
        regime_floor.value = state.regime_sigmoid["floor"]
        regime_ceiling.value = state.regime_sigmoid["ceiling"]
        regime_steep.value = state.regime_sigmoid["steepness"]
        regime_mid.value = state.regime_sigmoid["midpoint"]
        state.regime_x_grid = np.linspace(x_data.min() - 5, x_data.max() + 5, 300)
        state.regime_kernel_y = regime_kernel(x_data, y_data, state.regime_x_grid, regime_bandwidth.value)
        sig_p = [state.regime_sigmoid["floor"], state.regime_sigmoid["ceiling"], state.regime_sigmoid["steepness"], state.regime_sigmoid["midpoint"]]
        state.regime_sigmoid_y = sigmoid_val(state.regime_x_grid, sig_p)
        with regime_load_output:
            clear_output(wait=True)
            print(f"{len(state.regime_df)} days | floor={state.regime_sigmoid['floor']:.1f} ceil={state.regime_sigmoid['ceiling']:.1f}")
        update_regime_display()
    except Exception as e:
        with regime_load_output:
            clear_output(wait=True)
            print(f"Error: {e}")

def update_regime_display():
    if state.regime_df is None:
        return
    x_data, y_data = state.regime_df["SPX_DEV"].values, state.regime_df["CDX"].values
    state.regime_kernel_y = regime_kernel(x_data, y_data, state.regime_x_grid, regime_bandwidth.value)
    sig_p = [regime_floor.value, regime_ceiling.value, regime_steep.value, regime_mid.value]
    state.regime_sigmoid_y = sigmoid_val(state.regime_x_grid, sig_p)

    with regime_chart_output:
        clear_output(wait=True)
        fig = create_regime_chart(state.regime_df, state.regime_ols, state.regime_kernel_y, state.regime_sigmoid_y,
                                   state.regime_x_grid, regime_show_ols.value, regime_show_kernel.value, regime_show_sigmoid.value)
        display(fig)
    with regime_tables_output:
        clear_output(wait=True)
        sig_params = {"floor": regime_floor.value, "ceiling": regime_ceiling.value, "steepness": regime_steep.value, "midpoint": regime_mid.value}
        beta_df = calc_beta_table(x_data, y_data, state.regime_ols, sig_params, regime_bandwidth.value)
        stats_html = create_regime_stats_html(state.regime_df)
        beta_html = create_beta_table_html(beta_df)
        display(HTML(f'<div style="display:flex;flex-wrap:wrap;align-items:flex-start;">{stats_html}{beta_html}</div>'))

def on_regime_update(b):
    update_regime_display()
