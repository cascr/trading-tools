def load_data_bql(assets, field_map, start_date, end_date):
    all_dfs = []
    for asset in assets:
        ticker = ASSET_TICKERS.get(asset, asset)
        field = field_map.get(asset, "PX_LAST")
        if field == "PX_LAST":
            data_item = bq.data.px_last(dates=bq.func.range(start_date, end_date))
        else:
            data_item = bq.data.py026(dates=bq.func.range(start_date, end_date))
        request = bql.Request(ticker, {"VALUE": data_item})
        response = bq.execute(request)
        df = response.single().df()
        df = df.reset_index()
        df = df[["DATE", "VALUE"]].rename(columns={"VALUE": asset})
        df = df.set_index("DATE")
        all_dfs.append(df)
    result = pd.concat(all_dfs, axis=1)
    return result.sort_index().dropna()

def calculate_kmo(returns):
    corr = returns.corr().values
    try:
        inv_corr = np.linalg.inv(corr)
    except:
        inv_corr = np.linalg.inv(corr + 0.001 * np.eye(len(corr)))
    diag = np.diag(inv_corr)
    partial = -inv_corr / np.sqrt(np.outer(diag, diag))
    np.fill_diagonal(partial, 1.0)
    r_sq = corr ** 2
    p_sq = partial ** 2
    np.fill_diagonal(r_sq, 0)
    np.fill_diagonal(p_sq, 0)
    return r_sq.sum() / (r_sq.sum() + p_sq.sum())

def suggest_cohorts(returns, min_size=4, max_size=8, top_n=5):
    all_assets = list(returns.columns)
    results = []
    for size in range(min_size, min(max_size+1, len(all_assets)+1)):
        for combo in combinations(all_assets, size):
            assets = list(combo)
            n_credit = sum(1 for a in assets if a in CREDIT_ASSETS)
            n_equity = len(assets) - n_credit
            if n_credit >= 2 and n_equity >= 2:
                kmo = calculate_kmo(returns[assets])
                results.append({"assets": assets, "size": size, "kmo": kmo, "credit": n_credit, "equity": n_equity})
    results.sort(key=lambda x: x["kmo"], reverse=True)
    return results[:top_n]

def run_validation_tests(returns):
    results = {}
    kmo = calculate_kmo(returns)
    results["KMO"] = {"value": kmo, "status": "PASS" if kmo >= 0.8 else "WARN" if kmo >= 0.6 else "FAIL"}
    corr = returns.corr()
    mask = np.triu(np.ones_like(corr, dtype=bool), k=1)
    avg_corr = corr.where(mask).stack().abs().mean()
    results["Correlation"] = {"value": avg_corr, "status": "PASS" if avg_corr >= 0.3 else "FAIL"}
    results["corr_matrix"] = corr
    window = 60
    assets = returns.columns.tolist()
    stds = []
    for i in range(len(assets)):
        for j in range(i+1, len(assets)):
            roll_corr = returns[assets[i]].rolling(window=window).corr(returns[assets[j]]).dropna()
            stds.append(roll_corr.std())
    avg_std = np.mean(stds)
    results["Stability"] = {"value": avg_std, "status": "PASS" if avg_std < 0.2 else "WARN" if avg_std < 0.4 else "FAIL"}
    lag_corrs = []
    for i in range(len(assets)):
        for j in range(len(assets)):
            if i != j:
                lag_corr = returns[assets[i]].shift(1).corr(returns[assets[j]])
                lag_corrs.append(abs(lag_corr))
    avg_lag = np.mean(lag_corrs)
    results["LeadLag"] = {"value": avg_lag, "status": "PASS" if avg_lag < 0.1 else "WARN"}
    dist = 1 - corr.abs()
    condensed = squareform(dist.values)
    linkage_matrix = linkage(condensed, method="ward")
    labels = fcluster(linkage_matrix, t=linkage_matrix[:, 2].max() * 0.5, criterion="distance")
    n_clusters = len(set(labels))
    results["Clustering"] = {"value": n_clusters, "status": "PASS" if 2 <= n_clusters <= 4 else "WARN"}
    return results

def run_pca_analysis(returns, num_pcs=1):
    credit = [a for a in returns.columns if a in CREDIT_ASSETS]
    equity = [a for a in returns.columns if a in EQUITY_ASSETS]
    scaler = StandardScaler()
    data_scaled = scaler.fit_transform(returns)
    pca = PCA()
    pca_scores = pca.fit_transform(data_scaled)
    loadings = pd.DataFrame(pca.components_.T, columns=[f"PC{i+1}" for i in range(len(returns.columns))], index=returns.columns)
    var_explained = pd.Series(pca.explained_variance_ratio_, index=[f"PC{i+1}" for i in range(len(returns.columns))])
    expected = np.zeros_like(data_scaled)
    for i in range(num_pcs):
        beta = loadings[f"PC{i+1}"].values
        expected += np.outer(pca_scores[:, i], beta)
    residuals = data_scaled - expected
    z_scores = pd.DataFrame((residuals - residuals.mean(axis=0)) / residuals.std(axis=0), index=returns.index, columns=returns.columns)
    for asset in credit:
        if asset in z_scores.columns:
            z_scores[asset] = -z_scores[asset]
    return {"z_scores": z_scores, "loadings": loadings, "variance": var_explained, "credit": credit, "equity": equity}

def get_ranking(z_scores, credit, as_of_date=None):
    if as_of_date is None:
        as_of_date = z_scores.index[-1]
    z_row = z_scores.loc[as_of_date]
    ranking = []
    for asset in z_scores.columns:
        z = z_row[asset]
        if abs(z) >= 2:
            signal = "VERY RICH" if z > 0 else "VERY CHEAP"
        elif abs(z) >= 1:
            signal = "Rich" if z > 0 else "Cheap"
        else:
            signal = "Fair"
        ranking.append({"Asset": asset, "Type": "Credit" if asset in credit else "Equity", "Z-Score": z, "Signal": signal})
    df = pd.DataFrame(ranking).sort_values("Z-Score", ascending=False)
    df["Rank"] = range(1, len(df) + 1)
    return df[["Rank", "Asset", "Type", "Z-Score", "Signal"]]

def get_rv_opportunities(ranking):
    opps = {}
    for asset_type in ["Credit", "Equity"]:
        subset = ranking[ranking["Type"] == asset_type].sort_values("Z-Score", ascending=False)
        if len(subset) >= 2:
            rich = subset.iloc[0]
            cheap = subset.iloc[-1]
            opps[f"{asset_type} RV"] = {"Short": rich["Asset"], "Short Z": rich["Z-Score"], "Long": cheap["Asset"], "Long Z": cheap["Z-Score"], "Spread": rich["Z-Score"] - cheap["Z-Score"]}
    return opps

def get_changes(raw_data, asset):
    if asset in CREDIT_ASSETS:
        return raw_data[asset].diff().dropna()
    else:
        return raw_data[asset].pct_change().dropna() * 100

def create_kernel_regression_chart(raw_data, asset1, asset2, lookback=756):
    recent = raw_data.iloc[-lookback:]
    x_changes = get_changes(recent, asset1).values
    y_changes = get_changes(recent, asset2).values
    min_len = min(len(x_changes), len(y_changes))
    x_changes = x_changes[-min_len:]
    y_changes = y_changes[-min_len:]
    mask = ~(np.isnan(x_changes) | np.isnan(y_changes))
    x_clean = x_changes[mask]
    y_clean = y_changes[mask]
    slope, intercept, r_value, p_value, std_err = stats.linregress(x_clean, y_clean)
    beta = slope
    r_squared = r_value ** 2
    kr = KernelRidge(alpha=1.0, kernel='rbf', gamma=0.1)
    X_fit = x_clean.reshape(-1, 1)
    kr.fit(X_fit, y_clean)
    x_sorted_idx = np.argsort(x_clean)
    x_sorted = x_clean[x_sorted_idx]
    y_kr = kr.predict(x_sorted.reshape(-1, 1))
    x_scale = bqp.LinearScale()
    y_scale = bqp.LinearScale()
    colors = np.linspace(0.2, 1, len(x_clean))
    color_scale = bqp.ColorScale(colors=["#a8dadc", "#264653"])
    scatter = bqp.Scatter(x=x_clean, y=y_clean, scales={'x': x_scale, 'y': y_scale, 'color': color_scale}, color=colors, default_size=15, opacities=[0.6]*len(x_clean))
    kr_line = bqp.Lines(x=x_sorted, y=y_kr, scales={'x': x_scale, 'y': y_scale}, colors=[COLORS["rich"]], stroke_width=3)
    lin_y = intercept + slope * x_sorted
    lin_line = bqp.Lines(x=x_sorted, y=lin_y, scales={'x': x_scale, 'y': y_scale}, colors=[COLORS["primary"]], stroke_width=2, line_style="dashed")
    x_label = f"{asset1} {'Spread Chg (bps)' if asset1 in CREDIT_ASSETS else '% Chg'}"
    y_label = f"{asset2} {'Spread Chg (bps)' if asset2 in CREDIT_ASSETS else '% Chg'}"
    x_axis = bqp.Axis(scale=x_scale, label=x_label, grid_lines="none")
    y_axis = bqp.Axis(scale=y_scale, orientation="vertical", label=y_label, grid_lines="none")
    tooltip = bqp.Tooltip(fields=['x', 'y'], formats=['.2f', '.2f'])
    scatter.tooltip = tooltip
    fig = bqp.Figure(marks=[scatter, kr_line, lin_line], axes=[x_axis, y_axis], title=f"Kernel Regression: {asset1} vs {asset2}", fig_margin={"top": 50, "bottom": 60, "left": 70, "right": 20}, background_style={"fill": COLORS["background"]})
    fig.layout.width = "500px"
    fig.layout.height = "400px"
    return fig, beta, r_squared

def create_zscore_scatter(z_scores, asset1, asset2, lookback=756):
    recent = z_scores.iloc[-lookback:]
    x_data = recent[asset1].values
    y_data = recent[asset2].values
    x_scale = bqp.LinearScale(min=-4, max=4)
    y_scale = bqp.LinearScale(min=-4, max=4)
    colors = np.linspace(0.2, 1, len(x_data))
    color_scale = bqp.ColorScale(colors=["#a8dadc", "#264653"])
    scatter = bqp.Scatter(x=x_data, y=y_data, scales={'x': x_scale, 'y': y_scale, 'color': color_scale}, color=colors, default_size=15, opacities=[0.6]*len(x_data))
    current = bqp.Scatter(x=[x_data[-1]], y=[y_data[-1]], scales={'x': x_scale, 'y': y_scale}, colors=[COLORS["rich"]], default_size=80, marker="diamond")
    h_line = bqp.Lines(x=[-4, 4], y=[0, 0], scales={'x': x_scale, 'y': y_scale}, colors=[COLORS["neutral"]], line_style="dotted", opacities=[0.5])
    v_line = bqp.Lines(x=[0, 0], y=[-4, 4], scales={'x': x_scale, 'y': y_scale}, colors=[COLORS["neutral"]], line_style="dotted", opacities=[0.5])
    h2_pos = bqp.Lines(x=[-4, 4], y=[2, 2], scales={'x': x_scale, 'y': y_scale}, colors=[COLORS["threshold"]], line_style="dashed", opacities=[0.4])
    h2_neg = bqp.Lines(x=[-4, 4], y=[-2, -2], scales={'x': x_scale, 'y': y_scale}, colors=[COLORS["threshold"]], line_style="dashed", opacities=[0.4])
    v2_pos = bqp.Lines(x=[2, 2], y=[-4, 4], scales={'x': x_scale, 'y': y_scale}, colors=[COLORS["threshold"]], line_style="dashed", opacities=[0.4])
    v2_neg = bqp.Lines(x=[-2, -2], y=[-4, 4], scales={'x': x_scale, 'y': y_scale}, colors=[COLORS["threshold"]], line_style="dashed", opacities=[0.4])
    x_axis = bqp.Axis(scale=x_scale, label=f"{asset1} Z-Score", grid_lines="none")
    y_axis = bqp.Axis(scale=y_scale, orientation="vertical", label=f"{asset2} Z-Score", grid_lines="none")
    tooltip = bqp.Tooltip(fields=['x', 'y'], formats=['.2f', '.2f'])
    scatter.tooltip = tooltip
    current.tooltip = bqp.Tooltip(fields=['x', 'y'], formats=['.2f', '.2f'], labels=['Current X', 'Current Y'])
    fig = bqp.Figure(marks=[h2_pos, h2_neg, v2_pos, v2_neg, h_line, v_line, scatter, current], axes=[x_axis, y_axis], title=f"Z-Score: {asset1} vs {asset2}", fig_margin={"top": 50, "bottom": 60, "left": 70, "right": 20}, background_style={"fill": COLORS["background"]})
    fig.layout.width = "450px"
    fig.layout.height = "400px"
    return fig

def create_zscore_spread_chart(z_scores, asset1, asset2, lookback=756):
    recent = z_scores.iloc[-lookback:]
    spread = recent[asset1] - recent[asset2]
    dates = recent.index.tolist()
    x_scale = bqp.DateScale()
    y_scale = bqp.LinearScale()
    line = bqp.Lines(x=dates, y=spread.values, scales={'x': x_scale, 'y': y_scale}, colors=[COLORS["primary"]], stroke_width=1.5)
    zero_line = bqp.Lines(x=[dates[0], dates[-1]], y=[0, 0], scales={'x': x_scale, 'y': y_scale}, colors=[COLORS["neutral"]], line_style="dotted", opacities=[0.5])
    pos2_line = bqp.Lines(x=[dates[0], dates[-1]], y=[2, 2], scales={'x': x_scale, 'y': y_scale}, colors=[COLORS["threshold"]], line_style="dashed", opacities=[0.4])
    neg2_line = bqp.Lines(x=[dates[0], dates[-1]], y=[-2, -2], scales={'x': x_scale, 'y': y_scale}, colors=[COLORS["threshold"]], line_style="dashed", opacities=[0.4])
    x_axis = bqp.Axis(scale=x_scale, grid_lines="none")
    y_axis = bqp.Axis(scale=y_scale, orientation="vertical", label="Z-Score Spread", grid_lines="solid", grid_color="#e0e0e0")
    index_sel = bqp.interacts.IndexSelector(scale=x_scale, marks=[line])
    fig = bqp.Figure(marks=[pos2_line, neg2_line, zero_line, line], axes=[x_axis, y_axis], title=f"Z-Score Spread: {asset1} minus {asset2}", fig_margin={"top": 50, "bottom": 40, "left": 70, "right": 20}, background_style={"fill": COLORS["background"]}, interaction=index_sel)
    fig.layout.width = "100%"
    fig.layout.height = "280px"
    return fig

        self.returns = None
        self.pca_results = None
        self.test_results = None
        self.suggested_cohorts = None
        self.selected_assets = []
        self.field_map = {}

def update_selected_display():
    items = []
    for asset in state.selected_assets:
        field = state.field_map.get(asset, "PX_LAST")
        items.append(f"{asset} [{field}]")
    selected_list.options = items

def on_add(b):
    asset = available_list.value
    if asset and asset not in state.selected_assets:
        state.selected_assets.append(asset)
        state.field_map[asset] = field_select.value
        update_selected_display()

def on_remove(b):
    if selected_list.value:
        asset = selected_list.value.split(" [")[0]
        if asset in state.selected_assets:
            state.selected_assets.remove(asset)
            if asset in state.field_map:
                del state.field_map[asset]
            update_selected_display()

def on_load(b):
    with data_output:
        clear_output(wait=True)
        if len(state.selected_assets) < 3:
            print("Add at least 3 assets")
            return
        state.raw_data = load_data_bql(state.selected_assets, state.field_map, start_date.value.strftime("%Y-%m-%d"), end_date.value.strftime("%Y-%m-%d"))
        state.returns = state.raw_data.pct_change().dropna()
        print(f"Loaded {len(state.returns)} days for {len(state.selected_assets)} assets")
        pair_asset1.options = state.selected_assets
        pair_asset2.options = state.selected_assets
        if len(state.selected_assets) >= 2:
            pair_asset1.value = state.selected_assets[0]
            pair_asset2.value = state.selected_assets[1]

def on_run_tests(b):
    with tests_output:
        clear_output(wait=True)
        if state.returns is None:
            print("Load data first")
            return
        state.test_results = run_validation_tests(state.returns)
        for test, result in state.test_results.items():
            if test == "corr_matrix":
                continue
            color = "#2a9d8f" if result["status"] == "PASS" else "#e63946" if result["status"] == "FAIL" else "#f4a261"
            display(HTML(f"<span style='color:{color};font-weight:bold;'>{test}: {result['value']:.3f} [{result['status']}]</span>"))

def on_suggest(b):
    with cohort_output:
        clear_output(wait=True)
        if state.returns is None:
            print("Load data first")
            return
        print("Finding optimal cohorts...")
        state.suggested_cohorts = suggest_cohorts(state.returns)
        for i, c in enumerate(state.suggested_cohorts):
            display(HTML(f"<b>Option {i+1}:</b> KMO={c['kmo']:.3f} | {c['credit']}C/{c['equity']}E | {', '.join(c['assets'])}"))

def on_run_pca(b):
    with pca_output:
        clear_output(wait=True)
        if state.returns is None:
            print("Load data first")
            return
        state.pca_results = run_pca_analysis(state.returns, num_pcs=num_pcs.value)
        var = state.pca_results["variance"]
        print(f"PC1: {var.iloc[0]:.1%} | PC2: {var.iloc[1]:.1%} | PC3: {var.iloc[2]:.1%}")
    with results_output:
        clear_output(wait=True)
        z = state.pca_results["z_scores"]
        ranking = get_ranking(z, state.pca_results["credit"])
        display(ranking)
        opps = get_rv_opportunities(ranking)
        for trade, d in opps.items():
            display(HTML(f"<b>{trade}:</b> Short {d['Short']} ({d['Short Z']:+.2f}) / Long {d['Long']} ({d['Long Z']:+.2f}) | Spread: {d['Spread']:.2f}"))

def on_pair(b):
    with chart_output:
        clear_output(wait=True)
        if state.pca_results is None or state.raw_data is None:
            print("Run PCA first")
            return
        a1, a2 = pair_asset1.value, pair_asset2.value
        if not a1 or not a2:
            print("Select both assets")
            return
        z = state.pca_results["z_scores"]
        kr_fig, beta, r2 = create_kernel_regression_chart(state.raw_data, a1, a2)
        stats_html = HTML(f"<div style='background:#f8f9fa;padding:10px;border-radius:5px;margin:10px 0;'><b>Regression Stats:</b> Beta = {beta:.4f} | R² = {r2:.4f}</div>")
        zscore_scatter = create_zscore_scatter(z, a1, a2)
        spread_chart = create_zscore_spread_chart(z, a1, a2)
        display(widgets.HBox([kr_fig, zscore_scatter]))
        display(stats_html)
        display(spread_chart)
